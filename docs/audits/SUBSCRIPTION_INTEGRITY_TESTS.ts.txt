import { expect, it } from 'vitest'
import { createDemoHousehold } from './data/fixtures/demoHousehold'
import { validateFinanceData } from './data/validate'
import { addBill, markBillPaid, updateBill, billsForMonth, type BillFields } from './domain/finance/bills'
import { activityLog } from './domain/finance/history'
import { monthResult } from './domain/finance/spending'
import { applyCancel, applyPause, simulateSubscription } from './domain/finance/subscriptions'
import { validityOn } from './domain/finance/validity'
import { buildReport } from './report/report'
import type { FinanceData } from './domain/finance/types'

const fields: BillFields = { name: 'Audit Stream', category: 'Subscriptions', owner: 'shared', dueDay: 5, startMonth: '2026-01', endMonth: null, note: '', debtId: null }
function fresh() {
  const seed = createDemoHousehold()
  const data = { ...seed, bills: [], billMonths: [], income: [], expenses: [], transfers: [], debts: [], goals: [], goalMovements: [], balances: [] }
  const added = addBill(data, 'audit', fields, 12345)
  if (added.status !== 'ok') throw Error('setup')
  return added.data
}
const paid = (month: string) => markBillPaid(fresh(), 'audit', month, { amount: 12345, paidOn: '2026-02-01', paidFrom: null, interest: null }, '2026-02-01')
const pdf = (data: FinanceData, month: string) => buildReport(data, { period: { kind: 'month', month }, today: '2026-12-31', generatedOn: '2026-12-31', includeAmounts: true, isDemo: true }).toString()
function hidden(before: FinanceData, after: FinanceData, month: string) {
  expect(before.billMonths[0].paid).toBe(true)
  expect(after.billMonths).toEqual(before.billMonths)
  expect(activityLog(before).filter(e => e.type === 'bill')).toHaveLength(1)
  expect(activityLog(after).filter(e => e.type === 'bill')).toHaveLength(0)
  expect(monthResult(before, month, '2026-12-31').spending.total).toBe(12345)
  expect(monthResult(after, month, '2026-12-31')).toMatchObject({ spending: { total: 0 }, planned: { knownTotal: 0 }, cashFlow: { moneyOut: 0 }, result: 0 })
  expect(monthResult(before, month, '2026-12-31').result).toBe(-12345)
  expect(pdf(before, month)).toContain('123.45')
  expect(pdf(after, month)).not.toContain('123.45')
  expect(validateFinanceData(after).status).toBe('valid')
}
it('Edit Details accepts end before a recorded payment; History, totals and PDF lose it', () => {
  const before = paid('2026-04')
  const edit = updateBill(before, 'audit', { ...fields, endMonth: '2026-02' })
  expect(edit.status).toBe('ok')
  if (edit.status === 'ok') hidden(before, edit.data, '2026-04')
})
it('restored paid month inside pausedMonths passes validation and disappears', () => {
  const before = paid('2026-04')
  hidden(before, { ...before, bills: before.bills.map(b => ({ ...b, pausedMonths: ['2026-04'] })) }, '2026-04')
})
it('current cancellation hides a later prepaid month even with a fresh simulation', () => {
  const before = paid('2026-04')
  const sim = simulateSubscription(before, 'audit', 'cancel', '2026-02-01')!
  expect(sim.charges[0].month).toBe('2026-02')
  hidden(before, applyCancel(before, sim), '2026-04')
})
it('fresh pause skips paid months and preserves History, totals and PDF', () => {
  const before = paid('2026-04')
  const sim = simulateSubscription(before, 'audit', 'pause', '2026-02-01')!
  expect(sim.charges.map(c => c.month)).toEqual(['2026-02', '2026-03', '2026-05'])
  const after = applyPause(before, sim)
  expect(activityLog(after)).toEqual(activityLog(before))
  expect(monthResult(after, '2026-04', '2026-12-31')).toEqual(monthResult(before, '2026-04', '2026-12-31'))
  expect(pdf(after, '2026-04')).toBe(pdf(before, '2026-04'))
})
it('stale pause at domain level can hide a newly recorded payment', () => {
  const data = fresh()
  const sim = simulateSubscription(data, 'audit', 'pause', '2026-02-01')!
  const before = paid('2026-04')
  hidden(before, applyPause(before, sim), '2026-04')
})
it('cancel before first charge leaves no payment or planned charge but validity stays active in first month', () => {
  const data = fresh()
  const after = applyCancel(data, simulateSubscription(data, 'audit', 'cancel', '2025-12-01')!)
  expect(after.bills[0]).toMatchObject({ endMonth: '2026-01', pausedMonths: ['2026-01'] })
  expect(billsForMonth(after, '2026-01')).toEqual([])
  expect(activityLog(after).filter(e => e.type === 'bill')).toEqual([])
  expect(monthResult(after, '2026-01', '2026-12-31').spending.total).toBe(0)
  expect(validityOn(after.bills[0], '2026-01')).toBe('active')
  expect(pdf(after, '2026-01')).not.toContain('123.45')
})
it('removing a past end recreates intervening unpaid obligations', () => {
  const end = updateBill(fresh(), 'audit', { ...fields, endMonth: '2026-02' })
  if (end.status !== 'ok') throw Error('setup')
  const reopen = updateBill(end.data, 'audit', fields)
  if (reopen.status !== 'ok') throw Error('setup')
  expect(billsForMonth(end.data, '2026-03')).toEqual([])
  expect(monthResult(reopen.data, '2026-03', '2026-06-01')).toMatchObject({ planned: { leftToPay: 12345 }, spending: { total: 0 } })
})
